${question.category}

${question.question}

${question.focus}
` ) .join(''); } catch (error) { questionList.innerHTML = '

Unable to load questions. Please refresh the page.

'; console.error(error); } } function renderReport(data) { const strengths = data.strengths.length ? data.strengths : ['No standout strengths identified']; const opportunities = data.opportunities.length ? data.opportunities : ['No immediate gaps identified']; reportSection.innerHTML = `
${data.averageScore}

AI Recommendation

${data.recommendation}

${data.averageScore} Average score
${strengths.length} Strong areas

Strengths

    ${strengths.map((item) => `
  • ${item}
    • `).join('')}

Follow-up focus

    ${opportunities.map((item) => `
  • ${item}
    • `).join('')}
`; } submitButton.addEventListener('click', async () => { const answers = Array.from(document.querySelectorAll('.answer-box')).map((field) => ({ id: Number(field.dataset.questionId), answer: field.value.trim() })); const notAnswered = answers.filter((item) => !item.answer).length; if (notAnswered > 0) { alert('Please answer all interview questions before submitting.'); return; } submitButton.disabled = true; submitButton.textContent = 'Evaluating...'; try { const response = await fetch('/api/submit', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ answers }) }); const payload = await response.json(); renderReport(payload); } catch (error) { console.error(error); alert('There was a problem generating the evaluation.
